import {
  DEFAULT_MAX_FILE_SIZE_BYTES,
  FileUploadError,
} from './types';
import type { UploadFile } from './types';
import {
  type ImageCompressor,
  type ProcessedFile,
} from './optionalTypes';
import {
  assertAllowedExtension,
  getFileExtension,

} from './filePolicy';

const JPEG_MAGIC = [0xff, 0xd8, 0xff] as const;
const PNG_MAGIC = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a] as const;
const PDF_MAGIC = [0x25, 0x50, 0x44, 0x46, 0x2d] as const;
const MAX_IMAGE_PIXELS = 40_000_000;
const MAX_JPEG_HEADER_BYTES = 256 * 1024;
const MAX_JPEG_HEADER_SEGMENTS = 512;
const JPEG_SOF_MARKERS = new Set([
  0xc0, 0xc1, 0xc2, 0xc3, 0xc5, 0xc6, 0xc7,
  0xc9, 0xca, 0xcb, 0xcd, 0xce, 0xcf,
]);

const SUPPORTED_TYPES = {
  jpg: { magic: JPEG_MAGIC },
  jpeg: { magic: JPEG_MAGIC },
  png: { magic: PNG_MAGIC },
  pdf: { magic: PDF_MAGIC },
} as const;

type SupportedExtension = keyof typeof SUPPORTED_TYPES;

const GENERIC_STORED_MIME = 'application/octet-stream';
function getLegacyStorageType(extension: string | null): {
  mimeType: 'image/jpeg' | 'image/png' | 'application/pdf';
  extension: 'jpg' | 'png' | 'pdf';
} | null {
  switch (extension) {
    case 'jpg':
    case 'jpeg':
      return { mimeType: 'image/jpeg', extension: 'jpg' };
    case 'png':
      return { mimeType: 'image/png', extension: 'png' };
    case 'pdf':
      return { mimeType: 'application/pdf', extension: 'pdf' };
    default:
      return null;
  }
}

export interface PrepareFileOptions {
  maxFileSizeBytes?: number;
  imageCompressor?: ImageCompressor;
  allowedExtensions?: readonly string[];
}

function hasMagic(bytes: Uint8Array, expected: readonly number[]): boolean {
  return expected.every((value, index) => bytes[index] === value);
}

function readDeclaredImageDimensions(
  bytes: Uint8Array,
  mimeType: 'image/jpeg' | 'image/png',
): { width: number; height: number } | null {
  const view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
  if (mimeType === 'image/png') {
    if (
      bytes.length < 33
      || !hasMagic(bytes, PNG_MAGIC)
      || view.getUint32(8) !== 13
      || bytes[12] !== 0x49 || bytes[13] !== 0x48
      || bytes[14] !== 0x44 || bytes[15] !== 0x52
    ) return null;
    return { width: view.getUint32(16), height: view.getUint32(20) };
  }

  if (!hasMagic(bytes, JPEG_MAGIC)) return null;
  let offset = 2;
  for (let segment = 0; segment < MAX_JPEG_HEADER_SEGMENTS && offset < bytes.length; segment++) {
    if (bytes[offset] !== 0xff) return null;
    while (offset < bytes.length && bytes[offset] === 0xff) offset++;
    if (offset >= bytes.length) return null;
    const marker = bytes[offset++];
    if (marker === 0xda || marker === 0xd9) return null;
    if (marker === 0xd8 || marker === 0x01 || (marker >= 0xd0 && marker <= 0xd7)) continue;
    if (offset + 2 > bytes.length) return null;
    const length = view.getUint16(offset);
    if (length < 2 || offset + length > bytes.length) return null;
    if (JPEG_SOF_MARKERS.has(marker)) {
      if (length < 8) return null;
      return {
        width: view.getUint16(offset + 5),
        height: view.getUint16(offset + 3),
      };
    }
    offset += length;
  }
  return null;
}

async function assertSafeImageDimensions(
  source: Blob,
  mimeType: 'image/jpeg' | 'image/png',
  maxPixels = MAX_IMAGE_PIXELS,
): Promise<void> {
  const byteLimit = mimeType === 'image/png' ? 33 : MAX_JPEG_HEADER_BYTES;
  const bytes = new Uint8Array(await source.slice(0, byteLimit).arrayBuffer());
  const dimensions = readDeclaredImageDimensions(bytes, mimeType);
  const pixels = dimensions ? dimensions.width * dimensions.height : 0;
  const pixelLimit = Math.min(maxPixels, MAX_IMAGE_PIXELS);
  if (
    !dimensions
    || !Number.isSafeInteger(pixels)
    || pixels <= 0
    || !Number.isSafeInteger(pixelLimit)
    || pixelLimit <= 0
    || pixels > pixelLimit
  ) {
    throw new FileUploadError(
      'IMAGE_PROCESSING_FAILED',
      dimensions ? '图片像素尺寸超出安全处理范围。' : '无法在有界文件头中确认图片尺寸，上传已停止。',
      { recoverable: false },
    );
  }
}

function normalizeDisplayName(name: string): string {
  // eslint-disable-next-line no-control-regex -- Strip control characters from the display-only filename.
  const printable = name.replace(/[\u0000-\u001f\u007f]/g, '').trim();
  return (printable || 'upload').slice(0, 255);
}

function requireSubtleCrypto(): SubtleCrypto {
  const subtle = globalThis.crypto?.subtle;
  if (!subtle) {
    throw new FileUploadError(
      'CHECKSUM_UNAVAILABLE',
      '当前浏览器无法计算文件校验值，上传已停止。',
      { recoverable: false },
    );
  }
  return subtle;
}

export async function sha256Hex(bytes: ArrayBuffer): Promise<string> {
  const digest = await requireSubtleCrypto().digest('SHA-256', bytes);
  return Array.from(new Uint8Array(digest), value => value.toString(16).padStart(2, '0')).join('');
}

function canvasToBlob(
  canvas: HTMLCanvasElement,
  mimeType: 'image/jpeg' | 'image/png',
  jpegQuality: number,
): Promise<Blob> {
  return new Promise((resolve, reject) => {
    canvas.toBlob(
      blob => blob
        ? resolve(blob)
        : reject(new FileUploadError(
            'IMAGE_PROCESSING_FAILED',
            '图片压缩失败，上传已停止。',
            { recoverable: true },
          )),
      mimeType,
      mimeType === 'image/jpeg' ? jpegQuality : undefined,
    );
  });
}

async function loadBrowserImage(source: Blob): Promise<{
  width: number;
  height: number;
  draw(context: CanvasRenderingContext2D, width: number, height: number): void;
  close(): void;
}> {
  if (typeof createImageBitmap === 'function') {
    const bitmap = await createImageBitmap(source);
    return {
      width: bitmap.width,
      height: bitmap.height,
      draw: (context, width, height) => context.drawImage(bitmap, 0, 0, width, height),
      close: () => bitmap.close(),
    };
  }

  if (typeof Image === 'undefined' || typeof URL === 'undefined') {
    throw new FileUploadError(
      'IMAGE_PROCESSING_FAILED',
      '当前浏览器不支持可靠图片压缩，上传已停止。',
      { recoverable: false },
    );
  }

  const objectUrl = URL.createObjectURL(source);
  const image = new Image();
  try {
    await new Promise<void>((resolve, reject) => {
      image.onload = () => resolve();
      image.onerror = () => reject(new FileUploadError(
        'IMAGE_PROCESSING_FAILED',
        '图片无法解码，上传已停止。',
        { recoverable: false },
      ));
      image.src = objectUrl;
    });
    return {
      width: image.naturalWidth,
      height: image.naturalHeight,
      draw: (context, width, height) => context.drawImage(image, 0, 0, width, height),
      close: () => URL.revokeObjectURL(objectUrl),
    };
  } catch (error) {
    URL.revokeObjectURL(objectUrl);
    throw error;
  }
}

export const compressBrowserImage: ImageCompressor = async (source, options) => {
  // Validate declared dimensions before either browser decoder can allocate pixels.
  await assertSafeImageDimensions(source, options.mimeType, options.maxPixels);
  if (typeof document === 'undefined') {
    throw new FileUploadError(
      'IMAGE_PROCESSING_FAILED',
      '当前环境无法执行浏览器图片压缩，上传已停止。',
      { recoverable: false },
    );
  }

  let decoded: Awaited<ReturnType<typeof loadBrowserImage>> | null = null;
  try {
    decoded = await loadBrowserImage(source);
    const pixels = decoded.width * decoded.height;
    if (!Number.isSafeInteger(pixels) || pixels <= 0 || pixels > options.maxPixels) {
      throw new FileUploadError(
        'IMAGE_PROCESSING_FAILED',
        '图片像素尺寸超出安全处理范围。',
        { recoverable: false },
      );
    }
    if (Math.max(decoded.width, decoded.height) <= options.maxDimension) return null;

    const scale = options.maxDimension / Math.max(decoded.width, decoded.height);
    const width = Math.max(1, Math.round(decoded.width * scale));
    const height = Math.max(1, Math.round(decoded.height * scale));
    const canvas = document.createElement('canvas');
    canvas.width = width;
    canvas.height = height;
    const context = canvas.getContext('2d');
    if (!context) {
      throw new FileUploadError(
        'IMAGE_PROCESSING_FAILED',
        '无法初始化图片压缩画布，上传已停止。',
        { recoverable: true },
      );
    }
    decoded.draw(context, width, height);
    return canvasToBlob(canvas, options.mimeType, options.jpegQuality);
  } catch (error) {
    if (error instanceof FileUploadError) throw error;
    throw new FileUploadError(
      'IMAGE_PROCESSING_FAILED',
      '图片处理失败，上传已停止。',
      { recoverable: true, cause: error },
    );
  } finally {
    decoded?.close();
  }
};

export async function prepareFile(
  file: File,
  options: PrepareFileOptions = {},
): Promise<ProcessedFile> {
  const allowedExtensions = normalizeAllowedExtensions(options.allowedExtensions);
  const maxFileSizeBytes = options.maxFileSizeBytes ?? DEFAULT_MAX_FILE_SIZE_BYTES;
  if (!Number.isSafeInteger(maxFileSizeBytes) || maxFileSizeBytes <= 0) {
    throw new FileUploadError('INVALID_CONFIGURATION', '文件大小上限配置无效。');
  }
  if (file.size <= 0 || file.size > maxFileSizeBytes) {
    throw new FileUploadError(
      'FILE_TOO_LARGE',
      file.size <= 0 ? '不能上传空文件。' : '文件超过允许的大小上限。',
      { recoverable: false },
    );
  }

  const sourceExtension = getFileExtension(file.name);
  assertAllowedExtension(sourceExtension, allowedExtensions);
  const legacyType = getLegacyStorageType(sourceExtension);
  if (!legacyType) {
    const bytes = await file.arrayBuffer();
    return {
      body: file,
      bytes,
      displayName: normalizeDisplayName(file.name),
      extension: 'bin',
      mimeType: GENERIC_STORED_MIME,
      sizeBytes: file.size,
      checksumSha256: await sha256Hex(bytes),
      compressed: false,
    };
  }

  const sourceLegacyExtension = sourceExtension as SupportedExtension;
  const expected = SUPPORTED_TYPES[sourceLegacyExtension];
  const declaredMime = file.type.trim().toLowerCase();
  if (declaredMime !== legacyType.mimeType) {
    throw new FileUploadError(
      'INVALID_FILE_TYPE',
      '文件扩展名与浏览器声明的类型不一致。',
      { recoverable: false },
    );
  }

  const header = new Uint8Array(await file.slice(0, 8).arrayBuffer());
  if (!hasMagic(header, expected.magic)) {
    throw new FileUploadError(
      'INVALID_FILE_TYPE',
      '文件内容与扩展名或 MIME 类型不一致。',
      { recoverable: false },
    );
  }

  let body: Blob = file;
  let compressed = false;
  if (legacyType.mimeType === 'image/jpeg' || legacyType.mimeType === 'image/png') {
    await assertSafeImageDimensions(file, legacyType.mimeType);
    const compressor = options.imageCompressor ?? compressBrowserImage;
    const candidate = await compressor(file, {
      mimeType: legacyType.mimeType,
      maxDimension: 2560,
      maxPixels: MAX_IMAGE_PIXELS,
      jpegQuality: 0.82,
    });
    if (candidate && candidate.size > 0 && candidate.size < file.size) {
      const candidateType = candidate.type.trim().toLowerCase();
      const candidateHeader = new Uint8Array(await candidate.slice(0, 8).arrayBuffer());
      if (candidateType !== legacyType.mimeType || !hasMagic(candidateHeader, expected.magic)) {
        throw new FileUploadError(
          'IMAGE_PROCESSING_FAILED',
          '图片压缩结果格式无效，上传已停止。',
          { recoverable: true },
        );
      }
      body = candidate;
      compressed = true;
    }
  }

  const bytes = await body.arrayBuffer();
  return {
    body,
    bytes,
    displayName: normalizeDisplayName(file.name),
    extension: legacyType.extension,
    mimeType: legacyType.mimeType,
    sizeBytes: body.size,
    checksumSha256: await sha256Hex(bytes),
    compressed,
  };
}

/**
 * Convenience boundary for the core service. Image processing is opt-in: the
 * default entry never imports this file, and the returned value is a plain
 * UploadFile so callers do not need to carry checksum/processing internals.
 */
export async function prepareFileForUpload(
  file: File,
  options: PrepareFileOptions = {},
): Promise<UploadFile> {
  const processed = await prepareFile(file, options);
  return {
    name: processed.displayName,
    size: processed.sizeBytes,
    type: processed.mimeType,
    arrayBuffer: async () => processed.bytes,
  };
}
